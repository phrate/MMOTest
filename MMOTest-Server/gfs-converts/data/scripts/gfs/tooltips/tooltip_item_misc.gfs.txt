[client]
[tooltip]

drop_display = "Drop `item_name`"

    use_display = "Use `item_name` x`item_qty`"
    drop_display = "Drop `item_name` x`item_qty`"
    examine_display = "Examine `item_name` x`item_qty`"

actions = [
    use: "items/use_item.gfs", use_display, {"slot"=item_slot}
    drop: "items/drop_item.gfs", drop_display, {"slot"=item_slot}
    examine: "items/examine_item.gfs", examine_display, {"slot"=item_slot, "name"=item_name}
]
